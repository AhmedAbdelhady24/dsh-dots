# dsh-dots Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a DeepSeek Harness plugin that provides persistent, always-on personal agents ("dots") that work autonomously on a goal, wake on schedules, drive a real browser with a live take-over view, and ask before risky actions.

**Architecture:** Two npm packages — a host plugin `dsh-dots` (dot store, session runner, scheduler bridge, browser tool, approval integration, memory) and a web client plugin `dsh-client-ui-dots` (management panel, per-dot chat, live browser view). The host plugin composes existing DSH services (`dsh-goal`, `dsh-schedule`, `dsh-user-approval`, `dsh-tools`, `dsh-storage`, `dsh-credentials`) rather than reimplementing them.

**Tech Stack:** TypeScript, Cordis services (`@deepseek-ai/cordis`), Schemastery config schema, `@deepseek-ai/dsh-tools` (`defineTool`), Playwright (browser), vitest, tsdown (build), pnpm.

**Spec:** [docs/superpowers/specs/2026-10-01-dsh-dots-design.md](../specs/2026-10-01-dsh-dots-design.md)

## Global Constraints

- Node `>=22`; TypeScript; pnpm as the package manager.
- Host package name `dsh-dots`, client package name `dsh-client-ui-dots`; both `"type": "module"`.
- Every package that mounts into a profile declares `dsh.bundle.patch: "./cordis.patch.yml"` and an `apply(ctx, config)` export with `name`, `inject`, and a Schemastery `Config`.
- Dot `name` is trimmed, non-empty, `<= 120` chars (matches dsh-schedule's title bound).
- Schedule selectors map 1:1 to `dsh-schedule` selectors: `after_seconds`, `at`, `every_seconds` (>= 60s), `daily`, `weekly`, `cron` (five-field Vixie).
- Dots run only while the DSH host is running.
- v1 approval is a declarative `ask`/`allow` rule list (no review model).

import { randomUUID } from "node:crypto"

export type ScheduleSelector =
  | { kind: "after_seconds"; seconds: number }
  | { kind: "at"; instant: string }
  | { kind: "every_seconds"; seconds: number }
  | { kind: "daily"; time: string; timeZone: string }
  | { kind: "weekly"; time: string; timeZone: string; weekdays: number[] }
  | { kind: "cron"; expression: string; timeZone: string }

export interface ScheduleSpec { selector: ScheduleSelector; prompt: string }

export interface ApprovalRule {
  /** Tool action namespace, e.g. "browser.click", "email.send" */
  action: string
  policy: "ask" | "allow"
}

export interface Dot {
  id: string
  name: string
  goal: string
  schedule?: ScheduleSpec
  browser: { profileDir: string; viewport?: { width: number; height: number } }
  memory: { facts: string[]; skills: string[] }
  approvalRules: ApprovalRule[]
  createdAt: number
  updatedAt: number
}

export interface NewDot {
  name: string
  goal: string
  schedule?: ScheduleSpec
  browser?: Partial<Dot["browser"]>
  approvalRules?: ApprovalRule[]
}

export function validateDotName(name: string): string {
  const trimmed = name.trim()
  if (!trimmed) throw new TypeError("dot name must be non-empty")
  if (trimmed.length > 120) throw new TypeError("dot name must be at most 120 characters")
  return trimmed
}

export function normalizeDot(input: NewDot): Dot {
  const now = Date.now()
  const id = randomUUID()
  return {
    id,
    name: validateDotName(input.name),
    goal: input.goal.trim(),
    ...(input.schedule ? { schedule: input.schedule } : {}),
    browser: {
      profileDir: `dots-${id}`,
      ...(input.browser?.viewport ? { viewport: input.browser.viewport } : {}),
    },
    memory: { facts: [], skills: [] },
    approvalRules: input.approvalRules ?? [],
    createdAt: now,
    updatedAt: now,
  }
}
```

import { normalizeDot } from "./domain.js"
import type { Dot, NewDot } from "./domain.js"

export interface DotsData { dots: Dot[] }

export class DotStore {
  private readonly data: DotsData
  constructor(data: DotsData) { this.data = data }
  list(): Dot[] { return [...this.data.dots] }
  get(id: string): Dot | undefined { return this.data.dots.find((d) => d.id === id) }
  create(input: NewDot): Dot {
    const dot = normalizeDot(input)
    this.data.dots.push(dot)
    return dot
  }
  update(id: string, patch: Partial<Omit<Dot, "id" | "createdAt">>): Dot {
    const i = this.data.dots.findIndex((d) => d.id === id)
    if (i < 0) throw new Error(`dot not found: ${id}`)
    const name = patch.name !== undefined ? patch.name.trim() : this.data.dots[i].name
    if (!name || name.length > 120) throw new TypeError("dot name must be non-empty and at most 120 characters")
    const next: Dot = { ...this.data.dots[i], ...patch, name, updatedAt: Date.now() }
    this.data.dots[i] = next
    return next
  }
  remove(id: string): void {
    const i = this.data.dots.findIndex((d) => d.id === id)
    if (i < 0) throw new Error(`dot not found: ${id}`)
    this.data.dots.splice(i, 1)
  }
}
```

import type { Dot } from "./domain.js"

export function renderMemorySection(memory: Dot["memory"], dotName: string): string {
  const facts = memory.facts.length ? memory.facts.map((f) => `- ${f}`).join("\n") : ""
  const skills = memory.skills.length ? memory.skills.map((s) => `- ${s}`).join("\n") : ""
  if (!facts && !skills) return ""
  const parts: string[] = [`## Memory for dot "${dotName}"`]
  if (facts) parts.push("### Facts\n" + facts)
  if (skills) parts.push("### Skills\n" + skills)
  return parts.join("\n\n")
}
```

import { defineTool } from "@deepseek-ai/dsh-tools"
import type { BrowserDriver } from "./driver.js"

export type Tool = ReturnType<typeof defineTool>

export function defineBrowserTools(driver: BrowserDriver): Tool[] {
  return [
    defineTool({
      name: "browser.navigate",
      description: "Open a URL in the dot's browser.",
      parameters: { url: { type: "string", required: true, description: "Full URL" } },
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async (args) => JSON.stringify(await driver.navigate(args.url)),
    }),
    defineTool({
      name: "browser.screenshot",
      description: "Capture the current browser screen.",
      parameters: {},
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "image", source: { type: "base64", data: v, media_type: "image/png" } }] },
      execute: async () => (await driver.screenshot()).dataUrl,
    }),
    defineTool({
      name: "browser.extract",
      description: "Return readable text of the current page.",
      parameters: {},
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async () => JSON.stringify(await driver.extract()),
    }),
    defineTool({
      name: "browser.click",
      description: "Click an element by CSS selector or visible text.",
      parameters: { selector: { type: "string", required: true } },
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async (args) => JSON.stringify(await driver.click(args.selector)),
    }),
    defineTool({
      name: "browser.type",
      description: "Type text into a field.",
      parameters: { selector: { type: "string", required: true }, value: { type: "string", required: true } },
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async (args) => JSON.stringify(await driver.type(args.selector, args.value)),
    }),
    defineTool({
      name: "browser.take_over",
      description: "Pause and hand the browser to the user.",
      parameters: { reason: { type: "string" } },
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async (args) => { await driver.takeOver(args.reason ?? ""); return "ok" },
    }),
    defineTool({
      name: "browser.hand_back",
      description: "Return browser control to the dot.",
      parameters: {},
      output: { schema: { type: "string" }, render: (_a, v) => [{ type: "text", text: v }] },
      execute: async () => { await driver.handBack(); return "ok" },
    }),
  ]
}

import type { ScheduleSpec } from "./domain.js"

export function toScheduleCreate(spec: ScheduleSpec, title: string): Record<string, unknown> {
  const { selector, prompt } = spec
  const base = { title, prompt }
  switch (selector.kind) {
    case "after_seconds": return { ...base, after_seconds: selector.seconds }
    case "at": return { ...base, at: selector.instant }
    case "every_seconds": return { ...base, every_seconds: selector.seconds }
    case "daily": return { ...base, daily: { time: selector.time, time_zone: selector.timeZone } }
    case "weekly": return { ...base, weekly: { time: selector.time, time_zone: selector.timeZone, weekdays: selector.weekdays } }
    case "cron": return { ...base, cron: { expression: selector.expression, time_zone: selector.timeZone } }
  }
}
```

- [ ] **Step 3: Write `approval.ts`**

```ts
import type { ApprovalRule } from "./domain.js"

const RISKY_PREFIXES = ["browser.", "email.", "payment.", "post.", "send."]

export function policyFor(action: string, rules: ApprovalRule[]): "ask" | "allow" {
  const hit = rules.find((r) => r.action === action)
  if (hit) return hit.policy
  if (RISKY_PREFIXES.some((p) => action.startsWith(p))) return "ask"
  return "allow"
}
```
